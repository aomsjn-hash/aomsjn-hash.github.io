const SHEET_ID = '1vME8fdOtbDmAcoK-uZBdCDgcJ_mcz_J751GfcThvTQo'; const SHEET_NAME = 'PetData'; const PHOTO_FOLDER_ID = '1t4C1JwuyJ50Yu4bkiAeGnK3HQQj2QaGG'; // โฟลเดอร์ PetPhoto ใน Google Drive // ใช้เฉพาะตอนชีตว่างเปล่า (ไม่มีหัวคอลัมน์เลย) const DEFAULT_HEADERS = [ 'petName', 'species', 'age', 'weight', 'condition', 'allergy', 'other', 'photoUrl', 'username', 'phone', 'address' ]; // ทำให้ชื่อหัวคอลัมน์เทียบกันได้ เช่น "Photo URL" / "photoUrl" -> "photourl" function norm(s) { return String(s).toLowerCase().replace(/[^a-z]/g, ''); } // เบอร์โทร: เก็บเฉพาะตัวเลข และตัด 0 นำหน้า เพื่อเทียบกันได้ function phoneKey(p) { return String(p || '').replace(/\D/g, '').replace(/^0+/, ''); } function jsonOut(obj, callback) { const text = JSON.stringify(obj); if (callback && /^[\w.$]+$/.test(callback)) { // JSONP: ให้หน้าเว็บอ่านข้อมูลได้โดยไม่ติด CORS return ContentService.createTextOutput(callback + '(' + text + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(text).setMimeType(ContentService.MimeType.JSON); } // ----- อ่านข้อมูล: ?action=profile&username=...&phone=... ----- function doGet(e) { const p = (e && e.parameter) || {}; if (p.action === 'profile') { try { return jsonOut(getProfile(p.username, p.phone), p.callback); } catch (err) { return jsonOut({ ok: false, error: String(err) }, p.callback); } } return ContentService.createTextOutput('PetData API is running'); } function getProfile(username, phone) { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); if (!sheet || sheet.getLastRow() < 2) return { ok: true, owner: null, pets: [] }; const values = sheet.getDataRange().getDisplayValues(); const headers = values[0].map(norm); const col = function () { for (let i = 0; i < arguments.length; i++) { const idx = headers.indexOf(arguments[i]); if (idx !== -1) return idx; } return -1; }; const c = { petName: col('petname', 'petsname'), species: col('species'), age: col('age', 'ages'), weight: col('weight'), condition: col('condition', 'underlyingcondition'), allergy: col('allergy', 'drugallergy'), other: col('other'), photo: col('photourl', 'photo'), username: col('username'), phone: col('phone', 'phonenumber'), address: col('address') }; const get = (row, i) => (i === -1 ? '' : String(row[i] || '').trim()); const wantUser = String(username || '').trim().toLowerCase(); const wantPhone = phoneKey(phone); if (!wantUser || !wantPhone) return { ok: false, error: 'missing username or phone' }; const pets = []; let owner = null; for (let r = 1; r < values.length; r++) { const row = values[r]; if (get(row, c.username).toLowerCase() !== wantUser) continue; if (phoneKey(get(row, c.phone)) !== wantPhone) continue; if (!get(row, c.petName)) continue; const addr = get(row, c.address); owner = { username: get(row, c.username), phone: get(row, c.phone), address: addr && addr !== '-' ? addr : (owner ? owner.address : '') }; // แปลงลิงก์ไฟล์ใน Drive เป็นลิงก์รูปย่อที่แสดงในหน้าเว็บได้ let photo = ''; const url = get(row, c.photo); const m = url.match(/\/d\/([-\w]+)/) || url.match(/[?&]id=([-\w]+)/); if (m) photo = 'https://drive.google.com/thumbnail?id=' + m[1] + '&sz=w800'; pets.push({ name: get(row, c.petName), species: get(row, c.species), age: get(row, c.age), weight: get(row, c.weight), condition: get(row, c.condition), allergy: get(row, c.allergy), other: get(row, c.other), photo: photo }); } return { ok: true, owner: owner, pets: pets }; } // ----- บันทึกข้อมูลจากฟอร์ม ----- function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow(DEFAULT_HEADERS); sheet.getRange(1, 1, 1, DEFAULT_HEADERS.length).setFontWeight('bold'); sheet.setFrozenRows(1); } let photoUrl = ''; if (d.photo) { const blob = Utilities.newBlob( Utilities.base64Decode(d.photo), 'image/jpeg', (d.petName || 'pet') + '_' + Date.now() + '.jpg' ); const folder = DriveApp.getFolderById(PHOTO_FOLDER_ID); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoUrl = file.getUrl(); } // ข้อมูลที่ได้รับ จับคู่กับหัวคอลัมน์ในชีตตามชื่อ (ไม่สนลำดับคอลัมน์) const data = { timestamp: new Date(), petname: d.petName, species: d.species, age: d.age, ages: d.age, weight: d.weight, condition: d.condition, underlyingcondition: d.condition, allergy: d.allergy, drugallergy: d.allergy, other: d.other, photourl: photoUrl, photo: photoUrl, username: d.username, phone: d.phone, phonenumber: d.phone, address: d.address }; const lastCol = sheet.getLastColumn(); const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(norm); const row = headers.map(h => (data[h] !== undefined ? data[h] : '')); const rowNum = sheet.getLastRow() + 1; // ตั้งคอลัมน์เบอร์โทรเป็นข้อความ เพื่อให้เลข 0 ตัวแรกไม่หาย headers.forEach((h, i) => { if (h === 'phone' || h === 'phonenumber') { sheet.getRange(rowNum, i + 1).setNumberFormat('@'); } }); sheet.getRange(rowNum, 1, 1, headers.length).setValues([row]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // ใช้ทดสอบในหน้า Apps Script: เลือกฟังก์ชันนี้แล้วกด Run function testPost() { const fake = { postData: { contents: JSON.stringify({ username: 'test', phone: '0812345678', address: 'test address', petName: 'TestPet', species: 'Dog', age: '2 years', weight: '5 kg', condition: '-', allergy: '-', other: '-', photo: '' }) } }; Logger.log(doPost(fake).getContent()); } // ทดสอบการอ่านโปรไฟล์ (ดูผลใน Execution log) function testProfile() { Logger.log(JSON.stringify(getProfile('test', '0812345678'))); }
